// AJAX handler for newsletter signup
add_action('wp_ajax_thh_popup_signup', 'thh_popup_signup_handler');
add_action('wp_ajax_nopriv_thh_popup_signup', 'thh_popup_signup_handler');
function thh_popup_signup_handler() {
    $email = sanitize_email($_POST['email'] ?? '');
    $phone = sanitize_text_field($_POST['phone'] ?? '');
    if (!$email || !is_email($email)) {
        wp_send_json_error('Ongeldig e-mailadres');
    }
    global $wpdb;
    $table = $wpdb->prefix . 'thh_newsletter';
    // Create table if not exists
    $wpdb->query("CREATE TABLE IF NOT EXISTS {$table} (
        id INT AUTO_INCREMENT PRIMARY KEY,
        email VARCHAR(255) NOT NULL,
        phone VARCHAR(50) DEFAULT '',
        created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
        UNIQUE KEY unique_email (email)
    )");
    $result = $wpdb->replace($table, [
        'email' => $email,
        'phone' => $phone,
        'created_at' => current_time('mysql')
    ]);
    if ($result !== false) {
        wp_send_json_success('Aangemeld!');
    } else {
        wp_send_json_error('Fout bij opslaan');
    }
}

    (function() {
        if (localStorage.getItem('thh_popup_shown')) return;

        var overlay = document.getElementById('thhPopupOverlay');
        var closeBtn = document.getElementById('thhPopupClose');
        var submitBtn = document.getElementById('thhPopupSubmit');
        var emailInput = document.getElementById('thhPopupEmail');
        var phoneInput = document.getElementById('thhPopupPhone');
        var errorDiv = document.getElementById('thhPopupError');
        var formDiv = document.getElementById('thhPopupForm');
        var successDiv = document.getElementById('thhPopupSuccess');

        // Show after 4 seconds
        setTimeout(function() {
            overlay.classList.add('thh-popup-active');
        }, 4000);

        function closePopup() {
            overlay.classList.remove('thh-popup-active');
            localStorage.setItem('thh_popup_shown', '1');
        }

        closeBtn.addEventListener('click', closePopup);
        overlay.addEventListener('click', function(e) {
            if (e.target === overlay) closePopup();
        });
        document.addEventListener('keydown', function(e) {
            if (e.key === 'Escape') closePopup();
        });

        submitBtn.addEventListener('click', function() {
            var email = emailInput.value.trim();
            if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
                errorDiv.style.display = 'block';
                emailInput.style.borderColor = '#D4213D';
                return;
            }
            errorDiv.style.display = 'none';
            emailInput.style.borderColor = '#ddd';
            submitBtn.textContent = 'EVEN GEDULD...';
            submitBtn.disabled = true;

            var phone = phoneInput.value.trim();
            if (phone) phone = '+31' + phone;

            var formData = new FormData();
            formData.append('action', 'thh_popup_signup');
            formData.append('email', email);
            formData.append('phone', phone);

            fetch('<?php echo admin_url("admin-ajax.php"); ?>', {
                method: 'POST',
                body: formData
            })
            .then(function(r) { return r.json(); })
            .then(function(data) {
                formDiv.style.display = 'none';
                successDiv.style.display = 'block';
                localStorage.setItem('thh_popup_shown', '1');
            })
            .catch(function() {
                formDiv.style.display = 'none';
                successDiv.style.display = 'block';
                localStorage.setItem('thh_popup_shown', '1');
            });
        });

        emailInput.addEventListener('keypress', function(e) {
            if (e.key === 'Enter') submitBtn.click();
        });
    })();
