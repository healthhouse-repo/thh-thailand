    (function() {
        var STORAGE_KEY = 'sh_popup_shown';
        var DELAY_MS = 4000;
        if (localStorage.getItem(STORAGE_KEY)) return;
        setTimeout(function() {
            var popup = document.getElementById('sh-welcome-popup');
            if (!popup) return;
            popup.style.display = 'block';
            document.body.style.overflow = 'hidden';
        }, DELAY_MS);
        document.addEventListener('click', function(e) {
            if (e.target.classList.contains('sh-popup-overlay') ||
                e.target.classList.contains('sh-popup-close')) {
                closePopup();
            }
        });
        document.addEventListener('keydown', function(e) {
            if (e.key === 'Escape') closePopup();
        });
        function closePopup() {
            var popup = document.getElementById('sh-welcome-popup');
            if (popup) {
                popup.style.display = 'none';
                document.body.style.overflow = '';
                localStorage.setItem(STORAGE_KEY, '1');
            }
        }
        document.addEventListener('submit', function(e) {
            if (e.target.id !== 'sh-popup-form') return;
            e.preventDefault();
            var email = e.target.querySelector('input[name="email"]').value;
            var phone = e.target.querySelector('input[name="phone"]').value;
            if (!email) return;
            var fd = new FormData();
            fd.append('action', 'sh_newsletter_signup');
            fd.append('email', email);
            if (phone) fd.append('phone', '+31' + phone);
            fetch('/wp-admin/admin-ajax.php', { method: 'POST', body: fd });
            e.target.style.display = 'none';
            var content = e.target.closest('.sh-popup-content');
            var success = content.querySelector('.sh-popup-success');
            if (success) success.style.display = 'block';
            var disc = content.querySelector('.sh-popup-disclaimer');
            if (disc) disc.style.display = 'none';
            localStorage.setItem(STORAGE_KEY, '1');
            setTimeout(closePopup, 8000);
        });
    })();

// === NEWSLETTER SIGNUP HANDLER ===
add_action('wp_ajax_sh_newsletter_signup', 'sh_handle_newsletter');
add_action('wp_ajax_nopriv_sh_newsletter_signup', 'sh_handle_newsletter');
function sh_handle_newsletter() {
    $email = sanitize_email($_POST['email'] ?? '');
    if (!is_email($email)) {
        wp_send_json_error(['message' => 'Ongeldig e-mailadres']);
    }
    $phone = sanitize_text_field($_POST['phone'] ?? '');
    $signups = get_option('sh_newsletter_signups', []);
    if (!in_array($email, $signups)) {
        $signups[] = $email;
        update_option('sh_newsletter_signups', $signups);
    }
    $log = get_option('sh_newsletter_log', []);
    $log[] = ['email' => $email, 'phone' => $phone, 'date' => current_time('mysql'), 'source' => 'popup'];
    update_option('sh_newsletter_log', $log);
    wp_send_json_success(['message' => 'Bedankt voor je aanmelding!']);
}

// REST endpoint
add_action('rest_api_init', function() {
    register_rest_route('sh/v1', '/newsletter', [
        'methods' => 'GET',
        'callback' => function() {
            return rest_ensure_response(get_option('sh_newsletter_log', []));
        },
        'permission_callback' => function() {
            return current_user_can('manage_options');
        }
    ]);
});
